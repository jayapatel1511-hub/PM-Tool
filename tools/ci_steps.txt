# CI steps (§43.5, FR-DLV-11), run by `python3 tools/ci_tests.py steps <job>` from the repository root.
# One step per line: `<job>[,<job>...]: <command>`. Add ` [always]` after the job names to run a step
# after an earlier one failed. Each job uploads what its steps write under ci-artifacts/.
# A new browser suite or check adds a line here; services and permissions stay in .github/workflows/ci.yml.

# Backend build, shared by the test jobs.
domain,api-1,api-2,api-3,checks: dotnet restore Hub.slnx
domain,api-1,api-2,api-3: dotnet build Hub.slnx --no-restore -c Release

# Tests with coverage (rules engine ≥ 95 % branches, services ≥ 70 %); `gate` merges the shards' files.
# Each test job writes a .trx and `domain` lists every case, so `gate` can count them (FR-DLV-13).
domain: dotnet test Hub.slnx --no-build -c Release --filter "FullyQualifiedName~Hub.Tests.Domain." --collect:"XPlat Code Coverage" --logger "trx;LogFileName=$CI_JOB.trx" --results-directory ci-artifacts/coverage
domain: mkdir -p ci-artifacts && dotnet test Hub.slnx --no-build -c Release --list-tests > ci-artifacts/listed-tests.txt
api-1,api-2,api-3: dotnet test Hub.slnx --no-build -c Release --filter "$(python3 tools/ci_tests.py shard "$CI_JOB")" --collect:"XPlat Code Coverage" --logger "trx;LogFileName=$CI_JOB.trx" --results-directory ci-artifacts/coverage
gate: python3 tools/coverage_gate.py ci-artifacts-in
gate [always]: python3 tools/ci_tests.py count ci-artifacts-in --main ci-artifacts-main/test-counts.json --labels "${LABELS:-}" --event "${GITHUB_EVENT_NAME:-}" --out ci-artifacts/test-counts.json

checks: if [ "${GITHUB_EVENT_NAME:-}" = pull_request ]; then python3 tools/test_pr_checks.py && python3 tools/pr_checks.py size --base HEAD^1 --labels "${LABELS:-}"; fi
checks: python3 tools/test_backup_helpers.py && python3 tools/test_pilot_initializer.py && python3 tools/test_pilot_operations.py && python3 tools/test_pilot_pitr.py
checks: python3 -m unittest tools.test_social_activation
checks: python3 -m unittest tools/preview/test_fixture_api.py tools/preview/test_refresh_fixture_copy.py tools/preview/test_seed_resources.py tools/preview/test_seed_staffing.py scripts/test_extend_review_credentials.py scripts/test_seed_homedev_tuesday_review.py -v
checks: mkdir -p ci-artifacts && dotnet list Hub.slnx package --vulnerable --include-transitive | tee ci-artifacts/vuln.txt && ! grep -q "Critical" ci-artifacts/vuln.txt
